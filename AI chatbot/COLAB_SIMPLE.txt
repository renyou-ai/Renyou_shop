# Voici comment utiliser votre code RAG dans Colab

from langchain.document_loaders import PyPDFLoader, TextLoader
from langchain.text_splitter import CharacterTextSplitter
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.vectorstores import Chroma

# Utiliser les chemins Colab
PDF_FOLDER = '/content/drive/My Drive/RENYOU/PDFs'
TXT_FOLDER = '/content/drive/My Drive/RENYOU/TXTs'

# Charger les PDFs
documents = []

# Charger les PDFs
import os
for pdf_file in os.listdir(PDF_FOLDER):
    if pdf_file.endswith('.pdf'):
        try:
            loader = PyPDFLoader(os.path.join(PDF_FOLDER, pdf_file))
            docs = loader.load()
            documents.extend(docs)
            print(f"✅ Chargé: {pdf_file}")
        except Exception as e:
            print(f"❌ Erreur: {pdf_file}: {e}")

# Charger les TXTs
for txt_file in os.listdir(TXT_FOLDER):
    if txt_file.endswith('.txt'):
        try:
            loader = TextLoader(os.path.join(TXT_FOLDER, txt_file), encoding='utf-8')
            docs = loader.load()
            documents.extend(docs)
            print(f"✅ Chargé: {txt_file}")
        except Exception as e:
            print(f"❌ Erreur: {txt_file}: {e}")

print(f"\n✅ Total: {len(documents)} documents chargés")

# Traiter les documents
text_splitter = CharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
chunks = text_splitter.split_documents(documents)

print(f"✅ {len(chunks)} chunks créés")

# Créer les embeddings
embeddings = HuggingFaceEmbeddings()
vectorstore = Chroma.from_documents(chunks, embeddings)

print("✅ Vectorstore créé!")

CELLULE 6: RÉPONDRE À DES QUESTIONS
════════════════════════════════════════════════════════════════════════════════

from langchain.llms import HuggingFaceHub
from langchain.chains import RetrievalQA

# Créer une chaîne RAG
llm = HuggingFaceHub(
    repo_id="google/flan-t5-xxl",
    model_kwargs={"temperature": 0.7}
)

qa_chain = RetrievalQA.from_chain_type(
    llm=llm,
    chain_type="stuff",
    retriever=vectorstore.as_retriever()
)

# Poser une question
question = "Quelle est la maladie principale discutée?"
answer = qa_chain.run(question)

print(f"Question: {question}")
print(f"Réponse: {answer}")
